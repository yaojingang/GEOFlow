document.addEventListener('DOMContentLoaded',()=>{const small=window.matchMedia('(max-width:700px)').matches;document.querySelectorAll('[data-topic-auxiliary],[data-topic-reading-index]').forEach(el=>el.open=!small);document.querySelectorAll('[data-topic-copy]').forEach(button=>button.addEventListener('click',async()=>{const status=document.querySelector('[data-topic-copy-status]');try{await navigator.clipboard.writeText(button.dataset.url);status.textContent='链接已复制';}catch{status.textContent='复制未完成，可以复制浏览器地址栏中的链接。';}}));});
